cmake_minimum_required(VERSION 3.16)
project(quackshot_switch_remake_prototype VERSION 0.1 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(qs_core STATIC src/qs_game.c)
add_library(qs_trace STATIC src/qs_trace.c)
add_library(qs_motion STATIC src/qs_motion.c)
target_include_directories(qs_motion PUBLIC include)
target_include_directories(qs_trace PUBLIC include)
target_include_directories(qs_core PUBLIC include)
if(NOT MSVC)
    target_compile_options(qs_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(UNIX AND NOT APPLE)
    target_link_libraries(qs_core PUBLIC m)
endif()
enable_testing()
add_executable(qs_tests tests/test_game.c)
target_link_libraries(qs_tests PRIVATE qs_core)
add_test(NAME gameplay_core COMMAND qs_tests)
add_executable(qs_reference_city_tests tests/test_reference_city.c)
target_link_libraries(qs_reference_city_tests PRIVATE qs_core)
add_test(NAME reference_city_envelope COMMAND qs_reference_city_tests)
add_executable(qs_reference_obstacle_tests tests/test_reference_city_obstacle.c)
target_link_libraries(qs_reference_obstacle_tests PRIVATE qs_core)
add_test(NAME reference_city_obstacles COMMAND qs_reference_obstacle_tests)
add_executable(qs_trace_tests tests/test_trace.c)
target_link_libraries(qs_trace_tests PRIVATE qs_trace)
add_test(NAME runtime_trace COMMAND qs_trace_tests)
add_executable(qs_motion_tests tests/test_motion.c)
target_link_libraries(qs_motion_tests PRIVATE qs_motion qs_core)
add_test(NAME motion_trace COMMAND qs_motion_tests)
option(QS_BUILD_DESKTOP "Build SDL2 desktop prototype if SDL2 is installed" ON)
if(QS_BUILD_DESKTOP)
    find_package(SDL2 QUIET)
    if(SDL2_FOUND)
        add_executable(quackshot-prototype src/main_sdl.c)
        target_link_libraries(quackshot-prototype PRIVATE qs_core qs_trace qs_motion)
        if(TARGET SDL2::SDL2)
            target_link_libraries(quackshot-prototype PRIVATE SDL2::SDL2)
        else()
            target_include_directories(quackshot-prototype PRIVATE ${SDL2_INCLUDE_DIRS})
            target_link_libraries(quackshot-prototype PRIVATE ${SDL2_LIBRARIES})
        endif()
    else()
        message(STATUS "SDL2 not found: building core/tests only")
    endif()
endif()
