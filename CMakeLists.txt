cmake_minimum_required(VERSION 3.16)
project(quackshot_switch_remake_prototype VERSION 0.1 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(qs_core STATIC src/qs_game.c)
target_include_directories(qs_core PUBLIC include)
if(NOT MSVC)
    target_compile_options(qs_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(UNIX AND NOT APPLE)
    target_link_libraries(qs_core PUBLIC m)
endif()
enable_testing()
add_executable(qs_tests tests/test_game.c)
target_link_libraries(qs_tests PRIVATE qs_core)
add_test(NAME gameplay_core COMMAND qs_tests)
option(QS_BUILD_DESKTOP "Build SDL2 desktop prototype if SDL2 is installed" ON)
if(QS_BUILD_DESKTOP)
    find_package(SDL2 QUIET)
    if(SDL2_FOUND)
        add_executable(quackshot-prototype src/main_sdl.c)
        target_link_libraries(quackshot-prototype PRIVATE qs_core)
        if(TARGET SDL2::SDL2)
            target_link_libraries(quackshot-prototype PRIVATE SDL2::SDL2)
        else()
            target_include_directories(quackshot-prototype PRIVATE ${SDL2_INCLUDE_DIRS})
            target_link_libraries(quackshot-prototype PRIVATE ${SDL2_LIBRARIES})
        endif()
    else()
        message(STATUS "SDL2 not found: building core/tests only")
    endif()
endif()
